cmake_minimum_required(VERSION 3.22)
cmake_policy(SET CMP0091 NEW)
project(PilotSealflake VERSION 1.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "")
if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES)
  set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "")   # release builds pass "arm64;x86_64"
endif()

# Windows: link the C/C++ runtime statically, so the app and VST3 need no Visual C++ redistributable
# (and never pick up an older msvcp140.dll than the one they were built against)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# JUCE 8.0.10: a local ./JUCE checkout if there is one, otherwise downloaded at configure time
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/JUCE/CMakeLists.txt")
  add_subdirectory(JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()

# Plug-in formats per platform: AU on macOS, LV2 on Linux, VST3 + standalone everywhere
set(PILOT_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND PILOT_FORMATS AU)
elseif(UNIX)
  list(APPEND PILOT_FORMATS LV2)
endif()
# copy plug-ins into the user's plug-in folders after building (developer machines; CI turns it off)
if(APPLE)
  option(PILOT_COPY_PLUGINS "Copy plug-ins to the user plug-in folders after building" ON)
else()
  option(PILOT_COPY_PLUGINS "Copy plug-ins to the user plug-in folders after building" OFF)
endif()

juce_add_plugin(PilotSealflake
    COMPANY_NAME "Sealflake"
    BUNDLE_ID "com.sealflake.pilot"
    PRODUCT_NAME "Pilot Sealflake"
    PLUGIN_MANUFACTURER_CODE Slfk
    PLUGIN_CODE Pslf
    FORMATS ${PILOT_FORMATS}
    IS_SYNTH TRUE
    NEEDS_MIDI_INPUT TRUE
    NEEDS_MIDI_OUTPUT TRUE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS TRUE
    COPY_PLUGIN_AFTER_BUILD ${PILOT_COPY_PLUGINS}
    LV2URI "https://github.com/femifleming/orca_seaflake/pilot"
    VST3_CATEGORIES Instrument Synth
    AU_MAIN_TYPE kAudioUnitType_MusicDevice
    MICROPHONE_PERMISSION_ENABLED FALSE
    ICON_BIG "${CMAKE_CURRENT_SOURCE_DIR}/Resources/icon.png")

juce_add_binary_data(PilotFonts SOURCES
    Resources/fonts/input_mono_regular.ttf
    Resources/fonts/input_mono_medium.ttf)

# monome Teletype firmware (GPL-2.0, teletype/LICENSE) running inside Pilot: the real parser,
# ops, scripts, patterns and screens, plus the PI.* ops (teletype/src/ops/pilot.c) and a
# desktop host replacing the module's main.c (teletype/host/tt_host.c).
# teletype/src/match_token.c and scanner.c are generated by ragel; run tools/regen_teletype.sh after
# editing match_token.rl / scanner.rl.
file(GLOB TT_SOURCES CONFIGURE_DEPENDS
    teletype/src/*.c teletype/src/ops/*.c teletype/module/*.c teletype/host/*.c
    teletype/libavr32/src/*.c teletype/libavr32/src/euclidean/*.c)
add_library(TeletypeFirmware STATIC ${TT_SOURCES})
set_target_properties(TeletypeFirmware PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(TeletypeFirmware PRIVATE SIM)
if(MSVC)
  target_compile_options(TeletypeFirmware PRIVATE /w)
else()
  target_compile_options(TeletypeFirmware PRIVATE -w)
endif()
target_include_directories(TeletypeFirmware
    PUBLIC teletype/host
    PRIVATE teletype/shim teletype/src teletype/module teletype/libavr32/src)

target_sources(PilotSealflake PRIVATE
    Source/TeletypeHost.cpp
    Source/PilotEngine.cpp
    Source/Lang.cpp
    Source/PluginProcessor.cpp
    Source/PluginEditor.cpp
    Source/TeletypeView.cpp)

target_compile_definitions(PilotSealflake PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0
    JUCE_USE_CUSTOM_PLUGIN_STANDALONE_APP=1)

target_sources(PilotSealflake_Standalone PRIVATE Source/StandaloneApp.cpp)

target_link_libraries(PilotSealflake
    PRIVATE PilotFonts TeletypeFirmware juce::juce_audio_utils juce::juce_dsp juce::juce_audio_formats
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(NOT WIN32)   # clang-cl + link.exe (the Windows build) can't link JUCE's LTO flags
  target_link_libraries(PilotSealflake PUBLIC juce::juce_recommended_lto_flags)
endif()

# Offline render / measurement tool (not shipped)
juce_add_console_app(PilotRender PRODUCT_NAME "PilotRender")
target_sources(PilotRender PRIVATE tools/Render.cpp Source/PilotEngine.cpp Source/Lang.cpp Source/TeletypeHost.cpp)
target_compile_definitions(PilotRender PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
if(WIN32)   # the tests keep whole PilotEngines on the stack (~2 MB); Windows' default stack is 1 MB
  target_link_options(PilotRender PRIVATE /STACK:16777216)
endif()
target_link_libraries(PilotRender PRIVATE TeletypeFirmware juce::juce_dsp juce::juce_audio_formats juce::juce_audio_devices juce::juce_graphics PUBLIC juce::juce_recommended_config_flags)

# DAW-style host smoke test for the built VST3 (not shipped)
juce_add_console_app(PilotHostTest PRODUCT_NAME "PilotHostTest")
target_sources(PilotHostTest PRIVATE tools/HostTest.cpp)
target_compile_definitions(PilotHostTest PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_AU=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(PilotHostTest PRIVATE juce::juce_audio_processors juce::juce_gui_basics PUBLIC juce::juce_recommended_config_flags)
